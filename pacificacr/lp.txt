<script>
function updateStatus(){
  document.querySelectorAll("#jadwalTable tr[data-open]").forEach(row=>{
    let open = row.dataset.open;
    let close = row.dataset.close;
    let tz = row.dataset.tz;

    let now = new Date();
    let localTime = new Date(now.toLocaleString("en-US", {timeZone: tz}));

    let h = String(localTime.getHours()).padStart(2,'0');
    let m = String(localTime.getMinutes()).padStart(2,'0');
    let current = h + ":" + m;

    let statusCell = row.querySelector(".status");

    if(current >= open && current <= close){
      statusCell.innerHTML = '<span class="status-open">OPEN</span>';
    } else {
      statusCell.innerHTML = '<span class="status-close">CLOSED</span>';
    }
  });
}

updateStatus();
setInterval(updateStatus, 60000);
</script>
